Algorithm 3b: CreateCSpaceSubProblem

Input:
    Selected conflict c
    Robot set R′
    Conflict characteristics X
    Paths
    Global environment E
    Resolved_conflicts
    Resolution_attempts[c]

Output:
    Local environment E′
    Local queries Q′

1:  if Resolution_attempts[c] = ∅ then

2:      E′ ← PredictCSpaceRegion(
            c,
            R′,
            X,
            Resolved_conflicts,
            Paths)

3:  else

4:      Sprev ← MostRecentAttempt(
            Resolution_attempts[c])

5:      if Sprev.LocalEnvironment = E then

6:          return ∅

7:      end if

8:      E′ ← ExpandCSpace(
            c,
            R′,
            X,
            Resolved_conflicts,
            Paths,
            Sprev)

9:  end if

10: Q′ ← GenerateSubqueries(
        R′, E′, Paths, c)

11: return E′, Q′
